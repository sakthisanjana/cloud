async function searchBook() {
    const book = document.getElementById("bookName").value.trim();
    const results = document.getElementById("results");

    if (!book) {
        results.innerHTML = "<p class='error'>Enter a book name.</p>";
        return;
    }

    results.innerHTML = "<p>Searching books...</p>";

    try {
        const url =
            `https://openlibrary.org/search.json?q=${encodeURIComponent(book)}`;

        const response = await fetch(url);

        if (!response.ok) {
            throw new Error("API request failed");
        }

        const data = await response.json();
        displayBooks(data.docs);
    } catch (error) {
        results.innerHTML =
            "<p class='error'>Unable to retrieve books. Check your internet connection.</p>";
    }
}

function displayBooks(books) {
    const results = document.getElementById("results");
    results.innerHTML = "";

    if (!books || books.length === 0) {
        results.innerHTML = "<p>No books found.</p>";
        return;
    }

    books.slice(0, 10).forEach(book => {
        const title = book.title || "Unknown title";
        const authors = book.author_name
            ? book.author_name.join(", ")
            : "Unknown author";
        const year = book.first_publish_year || "Not available";

        const card = document.createElement("div");
        card.className = "book";

        const heading = document.createElement("h3");
        heading.textContent = title;

        const author = document.createElement("p");
        author.textContent = "Author: " + authors;

        const published = document.createElement("p");
        published.textContent = "Published: " + year;

        card.append(heading, author, published);
        results.appendChild(card);
    });
}

Step 6: Create src/AuthContext.js
This file manages the signed-in user and provides the login and logout functions.

import {
  createContext,
  useState,
  useEffect,
  useContext
} from "react";

import { auth, provider } from "./firebase";

import {
  onAuthStateChanged,
  signInWithPopup,
  signOut
} from "firebase/auth";

const AuthContext = createContext();

export const useAuth = () => useContext(AuthContext);

export function AuthProvider({ children }) {
  const [user, setUser] = useState(null);
  const [authLoading, setAuthLoading] = useState(true);

  useEffect(() => {
    const unsubscribe = onAuthStateChanged(
      auth,
      (currentUser) => {
        setUser(currentUser);
        setAuthLoading(false);
      }
    );

    return unsubscribe;
  }, []);

  const login = () => signInWithPopup(auth, provider);
  const logout = () => signOut(auth);

  return (
    <AuthContext.Provider
      value={{ user, login, logout, authLoading }}
    >
      {children}
    </AuthContext.Provider>
  );
}



Step 7: Create src/App.js
This is the main application. It includes Google login, adding text items, listing items, updating items, deleting items, and error messages. No images or Storage are used.

import { useEffect, useState } from "react";
import { db } from "./firebase";

import {
  collection,
  addDoc,
  getDocs,
  updateDoc,
  deleteDoc,
  doc,
  query,
  where
} from "firebase/firestore";

import { useAuth, AuthProvider } from "./AuthContext";

function App() {
  const { user, login, logout, authLoading } = useAuth();

  const [items, setItems] = useState([]);
  const [text, setText] = useState("");
  const [error, setError] = useState("");
  const [status, setStatus] = useState("");
  const [loading, setLoading] = useState(false);

  // READ: Load only the signed-in user's items
  const loadItems = async () => {
    if (!user) {
      setItems([]);
      return;
    }

    const itemsQuery = query(
      collection(db, "items"),
      where("owner", "==", user.uid)
    );

    const snapshot = await getDocs(itemsQuery);

    setItems(
      snapshot.docs.map((item) => ({
        id: item.id,
        ...item.data()
      }))
    );
  };

  useEffect(() => {
    if (user) {
      loadItems().catch((err) => {
        console.error(err);
        setError(
          (err.code ? err.code + ": " : "") + err.message
        );
      });
    } else {
      setItems([]);
      setError("");
      setStatus("");
    }
  }, [user]);

  // CREATE: Add a new item
  const addItem = async () => {
    setError("");
    setStatus("");

    if (!user) {
      setError("Please sign in first.");
      return;
    }

    if (!text.trim()) {
      setError("Please enter an item name.");
      return;
    }

    setLoading(true);

    try {
      await addDoc(collection(db, "items"), {
        text: text.trim(),
        owner: user.uid,
        createdAt: new Date().toISOString()
      });

      setText("");
      await loadItems();
      setStatus("Item added successfully!");
    } catch (err) {
      console.error("Add failed:", err);
      setError(
        (err.code ? err.code + ": " : "") + err.message
      );
    } finally {
      setLoading(false);
    }
  };

  // UPDATE: Change the item text
  const updateItem = async (id) => {
    const newText = window.prompt(
      "Enter the updated item name:"
    );

    if (!newText || !newText.trim()) return;

    setError("");
    setStatus("");

    try {
      await updateDoc(doc(db, "items", id), {
        text: newText.trim()
      });

      await loadItems();
      setStatus("Item updated successfully!");
    } catch (err) {
      console.error("Update failed:", err);
      setError(
        (err.code ? err.code + ": " : "") + err.message
      );
    }
  };

  // DELETE: Remove an item
  const deleteItem = async (id) => {
    if (!window.confirm("Delete this item?")) return;

    setError("");
    setStatus("");

    try {
      await deleteDoc(doc(db, "items", id));
      await loadItems();
      setStatus("Item deleted successfully!");
    } catch (err) {
      console.error("Delete failed:", err);
      setError(
        (err.code ? err.code + ": " : "") + err.message
      );
    }
  };

  const handleLogin = async () => {
    setError("");

    try {
      await login();
    } catch (err) {
      setError(err.message);
    }
  };

  const handleLogout = async () => {
    setError("");

    try {
      await logout();
    } catch (err) {
      setError(err.message);
    }
  };

  if (authLoading) {
    return <p>Checking login status...</p>;
  }

  return (
    <div style={{
      fontFamily: "Arial, sans-serif",
      padding: "30px",
      maxWidth: "700px"
    }}>
      <h1>React + Firebase CRUD App</h1>

      {user ? (
        <>
          <p>Welcome, {user.displayName}</p>

          <button onClick={handleLogout}>
            Sign out
          </button>

          <hr />

          <h2>Add Item</h2>

          <input
            type="text"
            placeholder="Enter item name"
            value={text}
            onChange={(e) => setText(e.target.value)}
          />

          <button
            onClick={addItem}
            disabled={loading}
            style={{ marginLeft: "10px" }}
          >
            {loading ? "Adding..." : "Add Item"}
          </button>

          {status && (
            <p style={{ color: "green" }}>{status}</p>
          )}

          {error && (
            <p style={{ color: "red" }}>
              Error: {error}
            </p>
          )}

          <h2>My Items</h2>

          {items.length === 0 ? (
            <p>No items found.</p>
          ) : (
            items.map((item) => (
              <div
                key={item.id}
                style={{
                  border: "1px solid #ccc",
                  borderRadius: "5px",
                  padding: "12px",
                  marginBottom: "10px"
                }}
              >
                <span>{item.text}</span>

                <button
                  onClick={() => updateItem(item.id)}
                  style={{ marginLeft: "15px" }}
                >
                  Update
                </button>

                <button
                  onClick={() => deleteItem(item.id)}
                  style={{ marginLeft: "8px" }}
                >
                  Delete
                </button>
              </div>
            ))
          )}
        </>
      ) : (
        <>
          <p>Please sign in to manage your items.</p>

          <button onClick={handleLogin}>
            Sign in with Google
          </button>

          {error && (
            <p style={{ color: "red" }}>{error}</p>
          )}
        </>
      )}
    </div>
  );
}

export default function RootApp() {
  return (
    <AuthProvider>
      <App />
    </AuthProvider>
  );
}



Step 8: Configure Firestore security rules
This step is essential. Without correct rules, you may see permission-denied.

In Firebase Console, go to Firestore Database → Rules. Replace the rules with the following and click Publish.

rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {
    match /items/{itemId} {

      allow create: if request.auth != null
                    && request.resource.data.owner
                       == request.auth.uid;

      allow read: if request.auth != null
                  && resource.data.owner
                     == request.auth.uid;

      allow update: if request.auth != null
                    && resource.data.owner
                       == request.auth.uid
                    && request.resource.data.owner
                       == request.auth.uid;

      allow delete: if request.auth != null
                    && resource.data.owner
                       == request.auth.uid;
    }
  }
}



These rules let users create items assigned to their own user ID and access only their own records. The query in App.js uses the same owner field, which is necessary for Firestore to evaluate the read permissions.

Step 7: Sample 2: Loop-level parallelism with reduction
Create the file:
nano loop_parallel.cpp
#include <omp.h>
#include <iostream>
#include <chrono>
using namespace std;
using namespace chrono;

long long serial_sum(int n) {
    long long sum = 0;
    for (int i = 1; i <= n; i++)
        sum += i;
    return sum;
}

long long parallel_sum(int n) {
    long long sum = 0;

    #pragma omp parallel for reduction(+:sum)
    for (int i = 1; i <= n; i++)
        sum += i;

    return sum;
}

int main() {
    int n = 10000000;

    auto start1 = high_resolution_clock::now();
    long long s1 = serial_sum(n);
    auto end1 = high_resolution_clock::now();

    auto start2 = high_resolution_clock::now();
    long long s2 = parallel_sum(n);
    auto end2 = high_resolution_clock::now();

    cout << "Serial Result: " << s1 << endl;
    cout << "Serial time: "
         << duration<double>(end1 - start1).count() << " seconds\n";

    cout << "Parallel Result: " << s2 << endl;
    cout << "Parallel time: "
         << duration<double>(end2 - start2).count() << " seconds\n";

    return 0;
}
